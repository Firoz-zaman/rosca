# Complete Supabase Database Setup Documentation

## Database Schema for ROSCA/Chitfund App

Here's everything you need to tell your employee to set up the same database:

---

## Step 1: Create Tables

Run this SQL in Supabase SQL Editor:

```sql
-- Enable UUID extension
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- 1. Profiles table (extends auth.users)
CREATE TABLE profiles (
  id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
  username TEXT UNIQUE,
  full_name TEXT,
  phone TEXT,
  role TEXT DEFAULT 'user' CHECK (role IN ('user', 'admin')),
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 2. ROSCAs/Groups table
CREATE TABLE roscas (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  name TEXT NOT NULL,
  description TEXT,
  contribution_amount NUMERIC NOT NULL,
  frequency TEXT NOT NULL CHECK (frequency IN ('daily', 'weekly', 'monthly')),
  total_slots INTEGER NOT NULL,
  status TEXT DEFAULT 'pending' CHECK (status IN ('pending', 'active', 'completed')),
  allocation_method TEXT DEFAULT 'random' CHECK (allocation_method IN ('random', 'banker', 'bidding')),
  created_by UUID REFERENCES auth.users(id) ON DELETE CASCADE,
  start_date DATE,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 3. ROSCA Members table
CREATE TABLE rosca_members (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  rosca_id UUID REFERENCES roscas(id) ON DELETE CASCADE,
  user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
  slot_number INTEGER,
  has_received BOOLEAN DEFAULT FALSE,
  joined_at TIMESTAMPTZ DEFAULT NOW(),
  UNIQUE(rosca_id, user_id)
);

-- Create indexes for performance
CREATE INDEX idx_profiles_role ON profiles(role);
CREATE INDEX idx_roscas_created_by ON roscas(created_by);
CREATE INDEX idx_rosca_members_user_id ON rosca_members(user_id);
CREATE INDEX idx_rosca_members_rosca_id ON rosca_members(rosca_id);
```

***

## Step 2: Enable Row Level Security (RLS)

```sql
-- Enable RLS on all tables
ALTER TABLE profiles ENABLE ROW LEVEL SECURITY;
ALTER TABLE roscas ENABLE ROW LEVEL SECURITY;
ALTER TABLE rosca_members ENABLE ROW LEVEL SECURITY;
```

## Step 3: Create RLS Policies

```sql
-- PROFILES POLICIES
CREATE POLICY "Public profiles viewable by everyone" ON profiles 
FOR SELECT USING (true);

CREATE POLICY "Users can update own profile" ON profiles 
FOR UPDATE USING (auth.uid() = id);

CREATE POLICY "Users can insert own profile" ON profiles 
FOR INSERT WITH CHECK (auth.uid() = id);

-- ROSCAS POLICIES
CREATE POLICY "Only admins can create ROSCAs" ON roscas 
FOR INSERT 
WITH CHECK (
  EXISTS (
    SELECT 1 FROM profiles 
    WHERE profiles.id = auth.uid() 
    AND profiles.role = 'admin'
  )
);

CREATE POLICY "Creators view their ROSCAs" ON roscas 
FOR SELECT 
USING (created_by = auth.uid());

CREATE POLICY "Creators can update their own ROSCAs" ON roscas 
FOR UPDATE 
USING (created_by = auth.uid());

-- ROSCA_MEMBERS POLICIES
CREATE POLICY "Allow viewing members" ON rosca_members 
FOR SELECT 
USING (true);

CREATE POLICY "ROSCA creators can add members" ON rosca_members 
FOR INSERT 
WITH CHECK (
  rosca_id IN (
    SELECT id FROM roscas WHERE created_by = auth.uid()
  )
);
```

## Step 4: Create Auto-Profile Trigger

This automatically creates a profile when a user signs up:

```sql
-- Function to auto-create profile on signup
CREATE OR REPLACE FUNCTION public.handle_new_user()
RETURNS TRIGGER AS $$
DECLARE
  user_count INTEGER;
BEGIN
  -- Count existing users
  SELECT COUNT(*) INTO user_count FROM auth.users;
  
  -- Insert profile with admin role if first user, otherwise regular user
  INSERT INTO public.profiles (id, full_name, role)
  VALUES (
    NEW.id, 
    NEW.raw_user_meta_data->>'full_name',
    CASE WHEN user_count <= 1 THEN 'admin' ELSE 'user' END
  );
  
  RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Trigger to auto-create profile
CREATE TRIGGER on_auth_user_created
  AFTER INSERT ON auth.users
  FOR EACH ROW EXECUTE FUNCTION public.handle_new_user();
```

***

## Step 5: Manually Assign Admin to Existing Users (If Needed)

If users signed up before the trigger was created:

```sql
-- Option 1: Make a specific user admin by email
INSERT INTO profiles (id, role, full_name)
SELECT 
  id, 
  'admin', 
  raw_user_meta_data->>'full_name'
FROM auth.users
WHERE email = 'admin@example.com'
ON CONFLICT (id) DO UPDATE SET role = 'admin';

-- Option 2: Make first user admin
INSERT INTO profiles (id, role, full_name)
SELECT 
  id, 
  'admin', 
  raw_user_meta_data->>'full_name'
FROM auth.users
ORDER BY created_at ASC
LIMIT 1
ON CONFLICT (id) DO UPDATE SET role = 'admin';
```

***

## How the System Works

### Role-Based Access Control:
- **Admin users** can create groups (ROSCAs)
- **Regular users** can only view groups they're added to
- First signup automatically becomes admin
- Additional admins can be assigned manually via SQL

### Ownership Model:
- Each admin can only **edit/delete groups they created**
- Admins cannot modify other admins' groups
- The `created_by` column tracks ownership

### Security:
- All policies are enforced at database level (RLS)
- Even if someone bypasses the UI, the database blocks unauthorized actions
- No recursion in policies (fixed for performance)

## Important Notes for Your Employee:

1. **Run all SQL in order** - tables must exist before policies
2. **First signup becomes admin automatically** - subsequent users are regular users
3. **Policies prevent infinite recursion** - we use simple, non-recursive checks
4. **created_by column** is critical - it determines who owns each group
5. **Test after setup** - Try creating a group as admin, verify regular users can't

This setup ensures secure, scalable role-based access control for your chitfund application!












-- ============================================
-- PAYMENT CYCLES TABLE
-- Tracks each rotation cycle in a ROSCA group
-- ============================================
CREATE TABLE payment_cycles (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  rosca_id UUID NOT NULL REFERENCES roscas(id) ON DELETE CASCADE,
  cycle_number INTEGER NOT NULL,
  
  -- Status workflow: bidding → payment → completed → overdue (if needed)
  status TEXT NOT NULL DEFAULT 'bidding' 
    CHECK (status IN ('bidding', 'payment', 'completed', 'overdue')),
  
  -- Timestamps for each phase
  bidding_start_date TIMESTAMPTZ NOT NULL,
  bidding_end_date TIMESTAMPTZ NOT NULL,
  payment_deadline_date TIMESTAMPTZ NOT NULL,
  cycle_end_date TIMESTAMPTZ NOT NULL,
  
  -- Winner (receiver) of this cycle
  winner_id UUID REFERENCES profiles(id),
  winning_bid_amount NUMERIC(10, 2),
  
  created_at TIMESTAMPTZ DEFAULT NOW(),
  
  -- Ensure one cycle per number per group
  UNIQUE(rosca_id, cycle_number)
);

-- Index for querying active cycles
CREATE INDEX idx_cycles_rosca_status ON payment_cycles(rosca_id, status);
CREATE INDEX idx_cycles_dates ON payment_cycles(bidding_end_date, payment_deadline_date);


-- ============================================
-- CYCLE BIDS TABLE
-- Stores all bids placed during bidding phase
-- ============================================
CREATE TABLE cycle_bids (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  cycle_id UUID NOT NULL REFERENCES payment_cycles(id) ON DELETE CASCADE,
  user_id UUID NOT NULL REFERENCES profiles(id) ON DELETE CASCADE,
  bid_amount NUMERIC(10, 2) NOT NULL,
  is_winner BOOLEAN DEFAULT FALSE,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Index for bid leaderboard queries
CREATE INDEX idx_bids_cycle ON cycle_bids(cycle_id, bid_amount);
CREATE INDEX idx_bids_user ON cycle_bids(user_id, cycle_id);


-- ============================================
-- CYCLE PAYMENTS TABLE
-- Tracks who paid and who verified for each cycle
-- ============================================
CREATE TABLE cycle_payments (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  cycle_id UUID NOT NULL REFERENCES payment_cycles(id) ON DELETE CASCADE,
  member_id UUID NOT NULL REFERENCES rosca_members(id) ON DELETE CASCADE,
  
  -- Payment tracking
  has_paid BOOLEAN DEFAULT FALSE,
  paid_at TIMESTAMPTZ,
  
  -- Verification tracking
  verified_by_receiver BOOLEAN DEFAULT FALSE,
  receiver_verified_at TIMESTAMPTZ,
  
  -- Admin override for late payments
  verified_by_admin BOOLEAN DEFAULT FALSE,
  admin_verified_at TIMESTAMPTZ,
  admin_id UUID REFERENCES profiles(id), -- Which admin verified
  
  created_at TIMESTAMPTZ DEFAULT NOW(),
  
  -- One payment record per member per cycle
  UNIQUE(cycle_id, member_id)
);

-- Index for payment status queries
CREATE INDEX idx_payments_cycle ON cycle_payments(cycle_id, has_paid, verified_by_receiver);
CREATE INDEX idx_payments_member ON cycle_payments(member_id);


-- ============================================
-- CYCLE ACTIVITIES TABLE
-- Activity feed for chat-style UI display
-- ============================================
CREATE TABLE cycle_activities (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  cycle_id UUID NOT NULL REFERENCES payment_cycles(id) ON DELETE CASCADE,
  
  activity_type TEXT NOT NULL CHECK (activity_type IN (
    'cycle_started',
    'bidding_opened',
    'bid_placed',
    'bidding_closed',
    'winner_declared',
    'payment_phase_started',
    'payment_made',
    'payment_verified',
    'cycle_completed',
    'admin_verified_payment'
  )),
  
  user_id UUID REFERENCES profiles(id), -- Who performed the action
  
  -- Flexible metadata storage (bid amounts, timestamps, etc.)
  metadata JSONB DEFAULT '{}',
  
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Index for activity feed queries
CREATE INDEX idx_activities_cycle ON cycle_activities(cycle_id, created_at DESC);


-- ============================================
-- RLS POLICIES
-- ============================================

-- Payment Cycles: Members can view cycles of groups they belong to
ALTER TABLE payment_cycles ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Members can view their group cycles"
  ON payment_cycles FOR SELECT
  USING (
    EXISTS (
      SELECT 1 FROM rosca_members
      WHERE rosca_members.rosca_id = payment_cycles.rosca_id
      AND rosca_members.user_id = auth.uid()
    )
  );

CREATE POLICY "System can manage cycles"
  ON payment_cycles FOR ALL
  USING (true)
  WITH CHECK (true);


-- Cycle Bids: Members can view and create bids for their group cycles
ALTER TABLE cycle_bids ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Members can view bids in their cycles"
  ON cycle_bids FOR SELECT
  USING (
    EXISTS (
      SELECT 1 FROM payment_cycles pc
      JOIN rosca_members rm ON rm.rosca_id = pc.rosca_id
      WHERE pc.id = cycle_bids.cycle_id
      AND rm.user_id = auth.uid()
    )
  );

CREATE POLICY "Members can create bids"
  ON cycle_bids FOR INSERT
  WITH CHECK (auth.uid() = user_id);


-- Cycle Payments: Members can view and update their own payments
ALTER TABLE cycle_payments ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Members can view payments in their cycles"
  ON cycle_payments FOR SELECT
  USING (
    EXISTS (
      SELECT 1 FROM payment_cycles pc
      JOIN rosca_members rm ON rm.rosca_id = pc.rosca_id
      WHERE pc.id = cycle_payments.cycle_id
      AND rm.user_id = auth.uid()
    )
  );

CREATE POLICY "Members can update their payment status"
  ON cycle_payments FOR UPDATE
  USING (
    EXISTS (
      SELECT 1 FROM rosca_members
      WHERE rosca_members.id = cycle_payments.member_id
      AND rosca_members.user_id = auth.uid()
    )
  );


-- Cycle Activities: Members can view activities in their group cycles
ALTER TABLE cycle_activities ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Members can view activities in their cycles"
  ON cycle_activities FOR SELECT
  USING (
    EXISTS (
      SELECT 1 FROM payment_cycles pc
      JOIN rosca_members rm ON rm.rosca_id = pc.rosca_id
      WHERE pc.id = cycle_activities.cycle_id
      AND rm.user_id = auth.uid()
    )
  );

CREATE POLICY "System can create activities"
  ON cycle_activities FOR INSERT
  WITH CHECK (true);
